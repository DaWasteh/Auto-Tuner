# =============================================================================
# Qwen3.8 DFlash2 — llama.cpp PR #27342, Windows x64 + Vulkan
# =============================================================================
# Stock llama.cpp b10590 supports first-generation DFlash, but NOT DFlash2's
# grouped local convolution + candidate selector. With an 81-tensor DFlash2
# GGUF it aborts after creating the old 58-tensor graph:
#
#   done_getting_tensors: wrong number of tensors; expected 81, got 58
#
# DFlash2 therefore needs its own checkout/build until PR #27342 is merged.
# Keep it BESIDE the normal b10590 build; do not overwrite the known-good
# mainline directory. AutoTuner discovers the output folder automatically.
#
# This recipe pins the exact PR revision validated with:
#   Target: I:\models\Alibaba\Qwen3.8\Qwen3.8-27B-UD-Q3_K_XL.gguf
#   Draft : I:\models\Alibaba\Qwen3.8\Qwen3.8-27B-DFlash2-Q4_K_M.gguf
#   Result: model loaded, /health ready, completion generated successfully.
#
# The pinned PR is based on an older mainline commit and reports build b10499.
# That is expected. AutoTuner recognizes the reviewed PR commit as DFlash2-
# capable even though its numeric build is lower than stock b10590.
# =============================================================================

param(
    [string]$Workspace = "L:\LAB\ai-local",
    [string]$Generator = "Visual Studio 18 2026",
    [ValidateRange(1, 256)]
    [int]$Parallel = 20
)

$ErrorActionPreference = "Stop"

function Invoke-NativeChecked {
    param(
        [Parameter(Mandatory = $true)][string]$Description,
        [Parameter(Mandatory = $true)][scriptblock]$Command
    )
    & $Command
    if ($LASTEXITCODE -ne 0) {
        throw "$Description failed with exit code $LASTEXITCODE"
    }
}

if (-not (Test-Path $Workspace -PathType Container)) {
    New-Item -ItemType Directory -Force -Path $Workspace | Out-Null
}
Set-Location (Resolve-Path $Workspace)

# Vulkan shader headers are kept outside the PR checkout so other llama.cpp
# builds can reuse them. A full install is cheap and avoids depending on a
# particular Vulkan SDK layout.
$spirv = Join-Path (Get-Location) "SPIRV-Headers"
if (-not (Test-Path "$spirv\.git")) {
    Invoke-NativeChecked "SPIRV-Headers clone" {
        git clone https://github.com/KhronosGroup/SPIRV-Headers.git $spirv
    }
} else {
    Invoke-NativeChecked "SPIRV-Headers update" {
        git -C $spirv pull --ff-only
    }
}

$spirvInstall = (Join-Path $spirv "install").Replace('\', '/')
Invoke-NativeChecked "SPIRV-Headers configure" {
    cmake -S $spirv -B "$spirv\build" `
        -G $Generator `
        -A x64 `
        "-DCMAKE_INSTALL_PREFIX=$spirvInstall"
}
Invoke-NativeChecked "SPIRV-Headers build" {
    cmake --build "$spirv\build" --config Release --parallel $Parallel
}
Invoke-NativeChecked "SPIRV-Headers install" {
    cmake --install "$spirv\build" --config Release
}

$repo = Join-Path (Get-Location) "pr27342_dflash2_llama.cpp"
$expectedCommit = "1deefcca395743049c3820ab8f9b15043f3e9446"

if (-not (Test-Path "$repo\.git")) {
    if (Test-Path $repo) {
        throw "Destination exists but is not a git checkout: $repo"
    }
    Invoke-NativeChecked "llama.cpp clone" {
        # Full history is required: llama.cpp embeds git rev-list --count HEAD
        # as its numeric compatibility build.
        git clone https://github.com/ggml-org/llama.cpp.git $repo
    }
}

$dirty = git -C $repo status --porcelain --untracked-files=no
if ($LASTEXITCODE -ne 0) {
    throw "Could not inspect DFlash2 checkout state"
}
if ($dirty) {
    throw "DFlash2 checkout has tracked local changes; refusing to replace them: $repo"
}

# Fetch to FETCH_HEAD and verify the reviewed revision before checkout. This
# prevents a mutable PR head from silently changing a reproducible build.
Invoke-NativeChecked "Fetch DFlash2 PR #27342" {
    git -C $repo fetch origin pull/27342/head --force
}
$fetchedCommit = (git -C $repo rev-parse FETCH_HEAD).Trim()
if ($LASTEXITCODE -ne 0) {
    throw "Could not resolve fetched DFlash2 PR revision"
}
if ($fetchedCommit -ne $expectedCommit) {
    throw "PR #27342 moved to $fetchedCommit; reviewed commit is $expectedCommit. Review/update this recipe before building a new revision."
}
Invoke-NativeChecked "Checkout reviewed DFlash2 revision" {
    git -C $repo checkout --detach $expectedCommit
}

$expectedBuildText = (git -C $repo rev-list --count HEAD).Trim()
if ($LASTEXITCODE -ne 0 -or $expectedBuildText -notmatch '^\d+$') {
    throw "Could not determine DFlash2 llama.cpp compatibility build"
}
$expectedBuild = [int]$expectedBuildText
if ($expectedBuild -lt 1000) {
    throw "History is incomplete; refusing suspicious build number $expectedBuild"
}
Write-Host "==> DFlash2 checkout: $repo"
Write-Host "==> Reviewed PR commit: $expectedCommit (embedded build b$expectedBuild)"

# Build the Web UI from this exact revision. llama-server remains usable as a
# local OpenAI-compatible server, not merely as a headless test executable.
Push-Location "$repo\tools\ui"
try {
    Invoke-NativeChecked "Web UI dependency install" { npm ci }
    Invoke-NativeChecked "Web UI build" { npm run build }
} finally {
    Pop-Location
}

# Preserve a previous known-good build until configure/build/smoke tests pass.
$buildDir = Join-Path $repo "build"
$backupDir = Join-Path $repo "build-backup-$PID"
if (Test-Path $backupDir) {
    throw "Backup directory already exists: $backupDir"
}
$hadBuild = Test-Path $buildDir
if ($hadBuild) {
    Move-Item $buildDir $backupDir
}

try {
    Invoke-NativeChecked "DFlash2 llama.cpp configure" {
        cmake -S $repo -B $buildDir `
            -G $Generator `
            -A x64 `
            -DLLAMA_BUILD_IS_DEV=ON `
            -DGGML_VULKAN=ON `
            -DGGML_NATIVE=OFF `
            -DGGML_AVX2=ON `
            -DGGML_FMA=ON `
            -DGGML_F16C=ON `
            -DBUILD_SHARED_LIBS=OFF `
            -DCMAKE_POLICY_DEFAULT_CMP0194=OLD `
            -DLLAMA_BUILD_SERVER=ON `
            -DLLAMA_BUILD_UI=ON `
            -DLLAMA_USE_PREBUILT_UI=OFF `
            -DLLAMA_CURL=OFF `
            -DGGML_CCACHE=OFF `
            -DGGML_VULKAN_CHECK_RESULTS=OFF `
            "-DCMAKE_PREFIX_PATH=$spirvInstall"
    }
    Invoke-NativeChecked "DFlash2 llama-server build" {
        cmake --build $buildDir --config Release `
            --target llama-server --parallel $Parallel
    }

    $server = Join-Path $buildDir "bin\Release\llama-server.exe"
    if (-not (Test-Path $server -PathType Leaf)) {
        throw "llama-server.exe was not produced: $server"
    }

    # llama-server writes version/help to stderr on some revisions. Route both
    # streams through cmd.exe so PowerShell does not turn valid output into a
    # terminating NativeCommandError.
    $versionOutput = (& $env:ComSpec /d /s /c "`"$server`" --version 2>&1" | Out-String).Trim()
    if ($LASTEXITCODE -ne 0) {
        throw "DFlash2 llama-server --version failed"
    }
    Write-Host $versionOutput
    $versionMatch = [regex]::Match(
        $versionOutput,
        '(?im)^\s*version:[^\r\n]*?\(\s*build\s+(\d+),\s*commit\s+([0-9a-f]+)'
    )
    if (-not $versionMatch.Success) {
        throw "llama-server --version did not report build + commit"
    }
    $reportedBuild = [int]$versionMatch.Groups[1].Value
    $reportedCommit = $versionMatch.Groups[2].Value.ToLowerInvariant()
    if ($reportedBuild -ne $expectedBuild) {
        throw "History/build mismatch: git count $expectedBuild, binary reports $reportedBuild"
    }
    if (-not $expectedCommit.StartsWith($reportedCommit) -and -not $reportedCommit.StartsWith($expectedCommit.Substring(0, 9))) {
        throw "Source/runtime commit mismatch: expected $expectedCommit, binary reports $reportedCommit"
    }

    $helpOutput = (& $env:ComSpec /d /s /c "`"$server`" --help 2>&1" | Out-String)
    if ($LASTEXITCODE -ne 0 -or $helpOutput -notmatch 'draft-dflash') {
        throw "Built server does not advertise --spec-type draft-dflash"
    }
} catch {
    if (Test-Path $buildDir) {
        Remove-Item $buildDir -Recurse -Force
    }
    if ($hadBuild -and (Test-Path $backupDir)) {
        Move-Item $backupDir $buildDir
    }
    throw
}

if ($hadBuild -and (Test-Path $backupDir)) {
    Remove-Item $backupDir -Recurse -Force
}

Write-Host ""
Write-Host "Success: $server"
Write-Host "Select 'pr27342_dflash2_llama.cpp' in AutoTuner for DFlash2."
Write-Host "Recommended Qwen3.8 DFlash2 arguments:"
Write-Host "  -md <DFlash2.gguf> --spec-type draft-dflash --spec-draft-n-max 7"
Write-Host "Stock b10590 remains available separately for every mainline feature."
