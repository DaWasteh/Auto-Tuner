# PowerShell build recipe for llama.cpp bNNNN pre-releases and development HEAD.
#
# Exact upstream bNNNN tags produce bNNNN_llama.cpp. A master commit that has
# not received its b-tag yet uses bNNNN_dev_COMMIT_llama.cpp so it cannot be
# mistaken for a published pre-release. Stable X.Y.Z releases are built by
# llama_stable_build.txt instead.

$ErrorActionPreference = "Stop"
Set-Location "L:\LAB\ai-local"

function Invoke-NativeChecked {
    param(
        [Parameter(Mandatory = $true)][string]$Description,
        [Parameter(Mandatory = $true)][scriptblock]$Command
    )
    & $Command
    if ($LASTEXITCODE -ne 0) {
        throw "$Description failed with exit code $LASTEXITCODE"
    }
}


function Get-ExactLlamaPrereleaseTag {
    param(
        [string]$Commit = "HEAD",
        [string]$Remote = "origin"
    )

    $sha = (git rev-parse $Commit 2>$null).Trim()
    if ($LASTEXITCODE -ne 0 -or -not $sha) {
        throw "Could not resolve llama.cpp commit '$Commit'"
    }

    # llama.cpp pushes the lightweight b-tag before publishing the GitHub
    # pre-release. Query the remote directly so that short publication-window
    # race is recognized without ever treating an older nearest tag as HEAD.
    $escapedSha = [regex]::Escape($sha)
    $remoteTags = git ls-remote --tags $Remote "refs/tags/b*" 2>$null
    foreach ($row in $remoteTags) {
        if ($row -match "^$escapedSha\s+refs/tags/(b\d+)(?:\^\{\})?$") {
            return $Matches[1]
        }
    }

    git fetch $Remote --tags --force 2>$null
    $pointingTags = @(
        git tag --points-at $Commit --list "b[0-9]*" 2>$null |
            Where-Object { $_ -match '^b\d+$' } |
            Sort-Object { [int]($_ -replace '^b', '') } -Descending
    )
    if ($pointingTags.Count -gt 0) { return $pointingTags[0] }

    return $null
}

$spirv = Join-Path (Get-Location) "SPIRV-Headers"
if (-not (Test-Path "$spirv\.git")) {
    Invoke-NativeChecked "SPIRV-Headers clone" {
        git clone https://github.com/KhronosGroup/SPIRV-Headers.git $spirv
    }
} else {
    Invoke-NativeChecked "SPIRV-Headers update" { git -C $spirv pull --ff-only }
}

Invoke-NativeChecked "SPIRV-Headers configure" {
    cmake -S $spirv -B "$spirv\build" `
        -G "Visual Studio 18 2026" `
        -A x64 `
        -DCMAKE_INSTALL_PREFIX="L:/LAB/ai-local/SPIRV-Headers/install"
}
Invoke-NativeChecked "SPIRV-Headers build" {
    cmake --build "$spirv\build" --config Release --parallel 20
}
Invoke-NativeChecked "SPIRV-Headers install" {
    cmake --install "$spirv\build" --config Release
}

# Build a fresh checkout without deleting the last known-good build. The
# process-specific staging name prevents two shells from clobbering each other.
$tmp = Join-Path (Get-Location) "_tmp_llama_$PID"
if (Test-Path $tmp) {
    throw "Staging directory already exists: $tmp"
}
Invoke-NativeChecked "llama.cpp clone" {
    git clone https://github.com/ggml-org/llama.cpp.git $tmp
}

Push-Location $tmp
try {
    $commit = (git rev-parse HEAD).Trim()
    if ($LASTEXITCODE -ne 0 -or -not $commit) {
        throw "Could not determine llama.cpp commit"
    }
    $shortCommit = $commit.Substring(0, 9)

    # Pre-release builds keep their bNNNN identity, while llama.cpp's runtime
    # also reports the current semantic development line
    # (for example 0.2.0-dev). Read the source of truth instead of pinning it.
    $cmakeText = Get-Content (Join-Path $tmp "CMakeLists.txt") -Raw
    $majorMatch = [regex]::Match($cmakeText, '(?m)^set\(LLAMA_VERSION_MAJOR\s+(\d+)\)')
    $minorMatch = [regex]::Match($cmakeText, '(?m)^set\(LLAMA_VERSION_MINOR\s+(\d+)\)')
    $patchMatch = [regex]::Match($cmakeText, '(?m)^set\(LLAMA_VERSION_PATCH\s+(\d+)\)')
    if (-not ($majorMatch.Success -and $minorMatch.Success -and $patchMatch.Success)) {
        throw "Could not determine llama.cpp semantic version from CMakeLists.txt"
    }
    $semanticVersion = "$($majorMatch.Groups[1].Value).$($minorMatch.Groups[1].Value).$($patchMatch.Groups[1].Value)"
    $expectedRuntimeVersion = "$semanticVersion-dev"

    # This is the same full-history count embedded by cmake/build-info.cmake.
    # It can legitimately be ahead of the latest published release while
    # master contains unreleased commits.
    $buildText = (git rev-list --count HEAD).Trim()
    if ($LASTEXITCODE -ne 0 -or $buildText -notmatch '^\d+$') {
        throw "Could not determine the llama.cpp compatibility build"
    }
    $expectedBuild = [int]$buildText
    if ($expectedBuild -lt 1000) {
        throw "History is incomplete; refusing suspicious build number $expectedBuild"
    }
    $buildTag = "b$expectedBuild"

    $exactTag = Get-ExactLlamaPrereleaseTag -Commit HEAD -Remote origin
    if ($exactTag -and $exactTag -ne $buildTag) {
        throw "Tag/history mismatch: exact tag $exactTag, commit count $buildTag"
    }
    $isPrerelease = $exactTag -eq $buildTag

    $description = (git describe --tags --always --match "b[0-9]*" 2>$null).Trim()
    if ($LASTEXITCODE -ne 0 -or -not $description) { $description = $shortCommit }
} finally {
    Pop-Location
}

# Exact pre-release tags keep the bNNNN_llama.cpp name. Untagged master
# commits get an explicit dev+commit suffix, so a binary reporting build 10548
# can never be mislabeled as the previous published b10545 pre-release.
$folderVersion = if ($isPrerelease) { $buildTag } else { "${buildTag}_dev_${shortCommit}" }
$dir = "${folderVersion}_llama.cpp"
$repo = Join-Path (Get-Location) $dir
if ((Test-Path $repo) -and $isPrerelease) {
    # Preserve an existing exact pre-release build without deleting it.
    $dir = "${buildTag}_${shortCommit}_llama.cpp"
    $repo = Join-Path (Get-Location) $dir
}
if (Test-Path $repo) {
    throw "Build directory already exists; nothing was removed: $repo"
}
Rename-Item $tmp $dir
$channel = if ($isPrerelease) { "exact pre-release tag $buildTag" } else { "development HEAD after $description" }
Write-Host "==> Build directory: $repo ($channel, commit $commit)"
Write-Host "==> Pre-release identity: $buildTag; runtime version: $expectedRuntimeVersion"
if (-not $isPrerelease) {
    Write-Host "NOTE: $buildTag is the embedded commit-count build, not yet an exact bNNNN pre-release tag."
}

Push-Location "$repo\tools\ui"
try {
    Invoke-NativeChecked "Web UI dependency install" { npm ci }
    Invoke-NativeChecked "Web UI build" { npm run build }
} finally {
    Pop-Location
}

Invoke-NativeChecked "llama.cpp configure" {
    cmake -S $repo -B "$repo\build" `
        -G "Visual Studio 18 2026" `
        -A x64 `
        -DLLAMA_BUILD_IS_DEV=ON `
        -DGGML_VULKAN=ON `
        -DGGML_NATIVE=OFF `
        -DGGML_AVX2=ON `
        -DGGML_FMA=ON `
        -DGGML_F16C=ON `
        -DBUILD_SHARED_LIBS=OFF `
        -DCMAKE_POLICY_DEFAULT_CMP0194=OLD `
        -DLLAMA_BUILD_SERVER=ON `
        -DLLAMA_BUILD_UI=ON `
        -DLLAMA_USE_PREBUILT_UI=OFF `
        -DLLAMA_CURL=OFF `
        -DGGML_CCACHE=OFF `
        -DGGML_VULKAN_CHECK_RESULTS=OFF `
        "-DCMAKE_PREFIX_PATH=L:/LAB/ai-local/SPIRV-Headers/install"
}
Invoke-NativeChecked "llama.cpp build" {
    cmake --build "$repo\build" --config Release --parallel 20
}
$server = "$repo\build\bin\Release\llama-server.exe"
# llama-server writes --version to stderr. Route both streams through cmd.exe
# so PowerShell's ErrorActionPreference=Stop does not turn valid output into a
# terminating NativeCommandError before we can inspect the exit code.
$versionOutput = (& $env:ComSpec /d /s /c "`"$server`" --version 2>&1" | Out-String).Trim()
$versionExit = $LASTEXITCODE
if ($versionExit -ne 0) {
    throw "llama-server smoke test failed with exit code $versionExit"
}
Write-Host $versionOutput
$versionMatch = [regex]::Match(
    $versionOutput,
    '(?im)^\s*version:\s*(\d+\.\d+\.\d+-dev)\s+\(\s*build\s+(\d+)\b'
)
if (-not $versionMatch.Success) {
    throw "llama-server --version did not report a semantic dev version plus numeric build"
}
$reportedVersion = $versionMatch.Groups[1].Value
$reportedBuild = [int]$versionMatch.Groups[2].Value
if ($reportedVersion -ne $expectedRuntimeVersion) {
    throw "Source/runtime version mismatch: expected $expectedRuntimeVersion, llama-server reports $reportedVersion"
}
if ($reportedBuild -ne $expectedBuild) {
    throw "History/build mismatch: git count $expectedBuild, llama-server reports $reportedBuild"
}

Write-Host ""
Write-Host "Success: $server ($reportedVersion, pre-release $buildTag, exact b-tag: $isPrerelease)"
Write-Host "OCR note: no CMake option controls Unlimited-OCR max_tiles."
Write-Host "Use llama.cpp b10287+ and an mmproj whose GGUF metadata contains:"
Write-Host "  clip.vision.preproc_max_tiles = 32"
