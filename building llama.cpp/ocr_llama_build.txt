# Legacy DeepSeek-OCR PR #17400 build.
# AutoTuner is server-based, so llama-server MUST be built. A build that only
# contains llama-mtmd-cli is intentionally not listed in the fork picker.
#
# Unlimited-OCR should use the current bNNNN pre-release build from
# llama_prerelease_build.txt
# instead: max_tiles=32 is projector GGUF metadata, not a CMake option.

$ErrorActionPreference = "Stop"
Set-Location "L:\LAB\ai-local"

function Invoke-NativeChecked {
    param(
        [Parameter(Mandatory = $true)][string]$Description,
        [Parameter(Mandatory = $true)][scriptblock]$Command
    )
    & $Command
    if ($LASTEXITCODE -ne 0) {
        throw "$Description failed with exit code $LASTEXITCODE"
    }
}

$repo = Join-Path (Get-Location) "ocr_b17400_llama.cpp"
if (-not (Test-Path "$repo\.git")) {
    Invoke-NativeChecked "llama.cpp clone" {
        git clone https://github.com/ggml-org/llama.cpp $repo
    }
}

$expectedCommit = "95cc5665859b49d7158c5c4abc9943adf109c6d5"
$dirty = git -C $repo status --porcelain --untracked-files=no
if ($LASTEXITCODE -ne 0) { throw "Could not inspect OCR checkout state" }
if ($dirty) {
    throw "OCR checkout has tracked local changes; refusing to reset: $repo"
}

# Fetch to FETCH_HEAD first: Git refuses to force-update a branch that is
# currently checked out in this worktree. The exact reviewed commit prevents
# a mutable PR head from silently changing this legacy build recipe.
Invoke-NativeChecked "Fetch DeepSeek-OCR PR #17400" {
    git -C $repo fetch origin pull/17400/head --force
}
$fetchedCommit = (git -C $repo rev-parse FETCH_HEAD).Trim()
if ($LASTEXITCODE -ne 0) { throw "Could not resolve fetched OCR commit" }
if ($fetchedCommit -ne $expectedCommit) {
    throw "PR #17400 moved to $fetchedCommit; reviewed commit is $expectedCommit"
}
Invoke-NativeChecked "Checkout reviewed DeepSeek-OCR commit" {
    git -C $repo checkout --detach $expectedCommit
}

# Preserve the last known-good build until configure, both target builds, and
# smoke tests all succeed. This also clears stale H:-drive CMake cache paths.
$buildDir = Join-Path $repo "build"
$backupDir = Join-Path $repo "build-backup-$PID"
if (Test-Path $backupDir) {
    throw "Backup directory already exists: $backupDir"
}
$hadBuild = Test-Path $buildDir
if ($hadBuild) {
    Move-Item $buildDir $backupDir
}

try {
    Invoke-NativeChecked "OCR fork configure" {
        cmake -S $repo -B $buildDir `
            -G "Visual Studio 18 2026" `
            -A x64 `
            -DGGML_VULKAN=ON `
            -DGGML_NATIVE=OFF `
            -DGGML_AVX2=ON `
            -DGGML_FMA=ON `
            -DGGML_F16C=ON `
            -DBUILD_SHARED_LIBS=OFF `
            -DLLAMA_BUILD_SERVER=ON `
            -DLLAMA_BUILD_UI=OFF `
            -DLLAMA_CURL=OFF `
            -DLLAMA_OPENSSL=OFF `
            -DCPPHTTPLIB_OPENSSL_SUPPORT=OFF `
            -DGGML_CCACHE=OFF `
            -DGGML_VULKAN_CHECK_RESULTS=OFF `
            -DCMAKE_POLICY_DEFAULT_CMP0194=OLD `
            "-DCMAKE_PREFIX_PATH=L:/LAB/ai-local/SPIRV-Headers/install"
    }
    Invoke-NativeChecked "OCR fork build" {
        cmake --build $buildDir --config Release `
            --target llama-server llama-mtmd-cli --parallel 20
    }
    Invoke-NativeChecked "OCR llama-server smoke test" {
        & "$buildDir\bin\Release\llama-server.exe" --version
    }
    Invoke-NativeChecked "OCR llama-mtmd-cli smoke test" {
        & "$buildDir\bin\Release\llama-mtmd-cli.exe" --help
    }
} catch {
    if (Test-Path $buildDir) {
        Remove-Item $buildDir -Recurse -Force
    }
    if ($hadBuild -and (Test-Path $backupDir)) {
        Move-Item $backupDir $buildDir
    }
    throw
}

if ($hadBuild -and (Test-Path $backupDir)) {
    Remove-Item $backupDir -Recurse -Force
}
Write-Host ""
Write-Host "AutoTuner can now discover: $repo"
Write-Host "This legacy PR build is for DeepSeek-OCR v1, not Unlimited-OCR 32 tiles."
