# =============================================================================
# Qwen3.8-Flash-Next (qwen4exp) — llama.cpp PR #27742
# Windows x64 + Vulkan
# =============================================================================
# Stock llama.cpp does not yet load the experimental qwen4exp architecture.
# This recipe builds Unsloth's PR branch at the exact revision tested here:
#
#   PR:     https://github.com/ggml-org/llama.cpp/pull/27742
#   Fork:   https://github.com/unslothai/llama.cpp
#   Branch: qwen4exp/qwen3.8-flash-next
#
# Keep this checkout beside the normal stable/pre-release build. AutoTuner
# discovers the resulting pr27742_qwen4exp_llama.cpp directory automatically.
# The PR is still under review, so the exact commit is pinned deliberately. If
# its head moves, this recipe stops instead of silently building unreviewed code.
# =============================================================================

param(
    [string]$Workspace = "L:\LAB\ai-local",
    [string]$Generator = "Visual Studio 18 2026",
    [ValidateRange(1, 256)]
    [int]$Parallel = 20
)

$ErrorActionPreference = "Stop"

function Invoke-NativeChecked {
    param(
        [Parameter(Mandatory = $true)][string]$Description,
        [Parameter(Mandatory = $true)][scriptblock]$Command
    )
    & $Command
    if ($LASTEXITCODE -ne 0) {
        throw "$Description failed with exit code $LASTEXITCODE"
    }
}

if (-not (Test-Path $Workspace -PathType Container)) {
    New-Item -ItemType Directory -Force -Path $Workspace | Out-Null
}
Set-Location (Resolve-Path $Workspace)

# Vulkan shader headers live outside the fork checkout so all local llama.cpp
# builds can share the same installation.
$spirv = Join-Path (Get-Location) "SPIRV-Headers"
if (-not (Test-Path "$spirv\.git")) {
    Invoke-NativeChecked "SPIRV-Headers clone" {
        git clone https://github.com/KhronosGroup/SPIRV-Headers.git $spirv
    }
} else {
    Invoke-NativeChecked "SPIRV-Headers update" {
        git -C $spirv pull --ff-only
    }
}

$spirvInstall = (Join-Path $spirv "install").Replace('\', '/')
Invoke-NativeChecked "SPIRV-Headers configure" {
    cmake -S $spirv -B "$spirv\build" `
        -G $Generator `
        -A x64 `
        "-DCMAKE_INSTALL_PREFIX=$spirvInstall"
}
Invoke-NativeChecked "SPIRV-Headers build" {
    cmake --build "$spirv\build" --config Release --parallel $Parallel
}
Invoke-NativeChecked "SPIRV-Headers install" {
    cmake --install "$spirv\build" --config Release
}

$repo = Join-Path (Get-Location) "pr27742_qwen4exp_llama.cpp"
$expectedCommit = "6c5afc86ae84448ae4d744e357017e2c490ad9c3"

if (-not (Test-Path "$repo\.git")) {
    if (Test-Path $repo) {
        throw "Destination exists but is not a git checkout: $repo"
    }
    Invoke-NativeChecked "llama.cpp clone" {
        # Full history is required because llama.cpp embeds the commit count as
        # its numeric compatibility build.
        git clone https://github.com/ggml-org/llama.cpp.git $repo
    }
}

$dirty = git -C $repo status --porcelain --untracked-files=no
if ($LASTEXITCODE -ne 0) {
    throw "Could not inspect qwen4exp checkout state"
}
if ($dirty) {
    throw "qwen4exp checkout has tracked local changes; refusing to replace them: $repo"
}

# Fetch the official PR ref, verify its immutable revision, then use a detached
# checkout. This avoids depending on a mutable fork branch after review.
Invoke-NativeChecked "Fetch Qwen3.8-Flash-Next PR #27742" {
    git -C $repo fetch origin pull/27742/head --force
}
$fetchedCommit = (git -C $repo rev-parse FETCH_HEAD).Trim()
if ($LASTEXITCODE -ne 0) {
    throw "Could not resolve fetched qwen4exp PR revision"
}
if ($fetchedCommit -ne $expectedCommit) {
    throw "PR #27742 moved to $fetchedCommit; reviewed commit is $expectedCommit. Review and update this recipe before building the new revision."
}
Invoke-NativeChecked "Checkout reviewed qwen4exp revision" {
    git -C $repo checkout --detach $expectedCommit
}

$expectedBuildText = (git -C $repo rev-list --count HEAD).Trim()
if ($LASTEXITCODE -ne 0 -or $expectedBuildText -notmatch '^\d+$') {
    throw "Could not determine qwen4exp llama.cpp compatibility build"
}
$expectedBuild = [int]$expectedBuildText
if ($expectedBuild -lt 1000) {
    throw "History is incomplete; refusing suspicious build number $expectedBuild"
}

$qwenSource = Join-Path $repo "src\models\qwen4exp.cpp"
if (-not (Test-Path $qwenSource -PathType Leaf)) {
    throw "Reviewed checkout does not contain the qwen4exp model implementation: $qwenSource"
}
$modelsCmake = Get-Content (Join-Path $repo "src\CMakeLists.txt") -Raw
$modelsCmakeRegistersQwen =
    $modelsCmake -match 'models/qwen4exp\.cpp' -or
    $modelsCmake -match 'GLOB\s+LLAMA_MODELS_SOURCES\s+"models/\*\.cpp"'
if (-not $modelsCmakeRegistersQwen) {
    throw "qwen4exp.cpp is not registered directly or through the models source glob"
}

Write-Host "==> Qwen3.8-Flash-Next checkout: $repo"
Write-Host "==> Reviewed PR commit: $expectedCommit (embedded build b$expectedBuild)"

# Build the Web UI from this exact revision when sources are available. The
# fallback keeps the recipe compatible if the PR is later rebased across a UI
# layout transition.
$uiSource = $null
foreach ($candidate in @("tools/ui", "tools/server/webui")) {
    $candidatePath = Join-Path $repo $candidate
    if (Test-Path (Join-Path $candidatePath "package.json") -PathType Leaf) {
        $uiSource = $candidatePath
        break
    }
}

if ($uiSource) {
    Write-Host "==> Building Web UI from $uiSource"
    Push-Location $uiSource
    try {
        if (Test-Path "package-lock.json" -PathType Leaf) {
            Invoke-NativeChecked "Web UI dependency install" { npm ci }
        } else {
            Invoke-NativeChecked "Web UI dependency install" { npm install }
        }
        Invoke-NativeChecked "Web UI build" { npm run build }
    } finally {
        Pop-Location
    }
    $usePrebuiltUi = "OFF"
} else {
    Write-Host "==> No Web UI sources found; using llama.cpp's prebuilt UI"
    $usePrebuiltUi = "ON"
}

# Preserve an existing known-good build until configure, compilation, and both
# smoke tests pass.
$buildDir = Join-Path $repo "build"
$backupDir = Join-Path $repo "build-backup-$PID"
if (Test-Path $backupDir) {
    throw "Backup directory already exists: $backupDir"
}
$hadBuild = Test-Path $buildDir
if ($hadBuild) {
    Move-Item $buildDir $backupDir
}

try {
    Invoke-NativeChecked "Qwen3.8-Flash-Next llama.cpp configure" {
        cmake -S $repo -B $buildDir `
            -G $Generator `
            -A x64 `
            -DLLAMA_BUILD_IS_DEV=ON `
            -DGGML_VULKAN=ON `
            -DGGML_NATIVE=OFF `
            -DGGML_AVX2=ON `
            -DGGML_FMA=ON `
            -DGGML_F16C=ON `
            -DBUILD_SHARED_LIBS=OFF `
            -DCMAKE_POLICY_DEFAULT_CMP0194=OLD `
            -DLLAMA_BUILD_SERVER=ON `
            -DLLAMA_BUILD_TESTS=ON `
            -DLLAMA_BUILD_UI=ON `
            "-DLLAMA_USE_PREBUILT_UI=$usePrebuiltUi" `
            -DLLAMA_CURL=OFF `
            -DGGML_CCACHE=OFF `
            -DGGML_VULKAN_CHECK_RESULTS=OFF `
            "-DCMAKE_PREFIX_PATH=$spirvInstall"
    }
    Invoke-NativeChecked "Qwen3.8-Flash-Next llama-server and architecture test build" {
        cmake --build $buildDir --config Release `
            --target llama-server test-llama-archs --parallel $Parallel
    }

    $server = Join-Path $buildDir "bin\Release\llama-server.exe"
    $archTest = Join-Path $buildDir "bin\Release\test-llama-archs.exe"
    if (-not (Test-Path $server -PathType Leaf)) {
        throw "llama-server.exe was not produced: $server"
    }
    if (-not (Test-Path $archTest -PathType Leaf)) {
        throw "test-llama-archs.exe was not produced: $archTest"
    }

    # llama.cpp writes version/test diagnostics to stderr on some revisions.
    # cmd.exe combines both streams without PowerShell converting valid output
    # into a terminating NativeCommandError.
    $versionOutput = (& $env:ComSpec /d /s /c "`"$server`" --version 2>&1" | Out-String).Trim()
    if ($LASTEXITCODE -ne 0) {
        throw "Qwen3.8-Flash-Next llama-server --version failed"
    }
    Write-Host $versionOutput
    $versionMatch = [regex]::Match(
        $versionOutput,
        '(?im)^\s*version:[^\r\n]*?\(\s*build\s+(\d+),\s*commit\s+([0-9a-f]+)'
    )
    if (-not $versionMatch.Success) {
        throw "llama-server --version did not report build and commit"
    }
    $reportedBuild = [int]$versionMatch.Groups[1].Value
    $reportedCommit = $versionMatch.Groups[2].Value.ToLowerInvariant()
    if ($reportedBuild -ne $expectedBuild) {
        throw "History/build mismatch: git count $expectedBuild, binary reports $reportedBuild"
    }
    if (-not $expectedCommit.StartsWith($reportedCommit)) {
        throw "Source/runtime commit mismatch: expected $expectedCommit, binary reports $reportedCommit"
    }

    $archOutput = (& $env:ComSpec /d /s /c "`"$archTest`" -a qwen4exp 2>&1" | Out-String).Trim()
    if ($LASTEXITCODE -ne 0) {
        Write-Host $archOutput
        throw "test-llama-archs failed for qwen4exp"
    }
    Write-Host $archOutput
} catch {
    if (Test-Path $buildDir) {
        Remove-Item $buildDir -Recurse -Force
    }
    if ($hadBuild -and (Test-Path $backupDir)) {
        Move-Item $backupDir $buildDir
    }
    throw
}

if ($hadBuild -and (Test-Path $backupDir)) {
    Remove-Item $backupDir -Recurse -Force
}

Write-Host ""
Write-Host "Success: $server"
Write-Host "qwen4exp architecture test passed for reviewed PR #27742 commit $expectedCommit."
Write-Host "Select 'pr27742_qwen4exp_llama.cpp' in AutoTuner for Qwen3.8-Flash-Next."
