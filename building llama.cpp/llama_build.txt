function Invoke-NativeChecked {
    param(
        [Parameter(Mandatory = $true)][string]$Description,
        [Parameter(Mandatory = $true)][scriptblock]$Command
    )
    & $Command
    if ($LASTEXITCODE -ne 0) {
        throw "$Description failed with exit code $LASTEXITCODE"
    }
}


function Get-LlamaBuildTag {
    param(
        [string]$Commit = "HEAD",
        [string]$Remote = "origin"
    )

    $sha = (git rev-parse $Commit 2>$null).Trim()
    if ($LASTEXITCODE -eq 0 -and $sha) {
        # llama.cpp b10470+ creates/pushes the lightweight release tag explicitly
        # in CI, before the GitHub Release is created. Ask the remote tag
        # namespace directly first, so even a just-created tag is recognized.
        $remoteTags = git ls-remote --tags $Remote "refs/tags/b*" 2>$null
        foreach ($row in $remoteTags) {
            if ($row -match "^$sha\s+refs/tags/(b\d+)$") { return $Matches[1] }
        }
    }

    git fetch $Remote --tags --force 2>$null

    $pointingTags = @(
        git tag --points-at $Commit --list "b[0-9]*" 2>$null |
            Where-Object { $_ -match '^b\d+$' } |
            Sort-Object { [int]($_ -replace '^b', '') } -Descending
    )
    if ($pointingTags.Count -gt 0) { return $pointingTags[0] }

    $desc = (git describe --tags --abbrev=0 --match "b[0-9]*" $Commit 2>$null).Trim()
    if ($LASTEXITCODE -eq 0 -and $desc -match 'b\d+') { return $Matches[0] }

    return "bUNKNOWN"
}

# Build a fresh checkout without deleting the last known-good build. The
# process-specific staging name prevents two shells from clobbering each other.
$tmp = Join-Path (Get-Location) "_tmp_llama_$PID"
if (Test-Path $tmp) {
    throw "Staging directory already exists: $tmp"
}
Invoke-NativeChecked "llama.cpp clone" {
    git clone https://github.com/ggml-org/llama.cpp.git $tmp
}

Push-Location $tmp
try {
    $version = Get-LlamaBuildTag -Commit HEAD -Remote origin
    $description = (git describe --tags --always --match "b[0-9]*" 2>$null).Trim()
    if ($LASTEXITCODE -ne 0 -or -not $description) { $description = $version }
    $commit = (git rev-parse HEAD).Trim()
    if ($LASTEXITCODE -ne 0) { throw "Could not determine llama.cpp commit" }
    $shortCommit = $commit.Substring(0, 9)
} finally {
    Pop-Location
}

$dir = "${version}_llama.cpp"
$repo = Join-Path (Get-Location) $dir
if (Test-Path $repo) {
    # Preserve the existing versioned build and publish this exact newer HEAD
    # under a commit-qualified folder that AutoTuner also discovers.
    $dir = "${version}_${shortCommit}_llama.cpp"
    $repo = Join-Path (Get-Location) $dir
}
if (Test-Path $repo) {
    throw "Build directory already exists; nothing was removed: $repo"
}
Rename-Item $tmp $dir
Write-Host "==> Build directory: $repo ($description, commit $commit)"

Push-Location "$repo\tools\ui"
try {
    Invoke-NativeChecked "Web UI dependency install" { npm ci }
    Invoke-NativeChecked "Web UI build" { npm run build }
} finally {
    Pop-Location
}

Invoke-NativeChecked "llama.cpp configure" {
    cmake -S $repo -B "$repo\build" `
        -G "Visual Studio 18 2026" `
        -A x64 `
        -DGGML_VULKAN=ON `
        -DGGML_NATIVE=OFF `

        -DLLAMA_BUILD_SERVER=ON `
        -DLLAMA_BUILD_UI=ON `
        -DLLAMA_USE_PREBUILT_UI=OFF `
        -DLLAMA_CURL=OFF `
        -DGGML_CCACHE=OFF `
        -DGGML_VULKAN_CHECK_RESULTS=OFF `
        "-DCMAKE_PREFIX_PATH=L:/LAB/ai-local/SPIRV-Headers/install"
}
Invoke-NativeChecked "llama.cpp build" {
    cmake --build "$repo\build" --config Release --parallel 20
}
Invoke-NativeChecked "llama-server smoke test" {
    & "$repo\build\bin\Release\llama-server.exe" --version
}
